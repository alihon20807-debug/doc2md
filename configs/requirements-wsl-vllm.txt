# Pinned versions for the WSL2 vLLM serving environment (~/.python312/ inside
# WSL, NOT this repo's own uv-managed venv - that venv only needs doc2md's
# own deps from pyproject.toml, it never talks to vllm/transformers
# directly, only over HTTP).
#
# Nothing installs from this file automatically today - it exists so the
# critical pin below is a real, git-tracked fact instead of only living in
# CLAUDE.md prose and handoffs/vllm_guide_handoff.md section 4.4. If the WSL
# venv is ever rebuilt from scratch, install these exact versions before
# anything else.
#
# transformers must be pinned to exactly this version, not "latest" and not
# left unpinned. transformers 5.15.0 introduced a stricter
# HeterogeneousConfigMixin guard on per-layer config attributes that
# conflicts with how vLLM 0.27.x reads Gemma 4's heterogeneous per-layer
# config (mixed head_dim=256 sliding-attention / head_dim=512 full-attention
# layers), raising AmbiguousGlobalPerLayerAttributeError at server startup.
# Confirmed upstream: https://github.com/vllm-project/vllm/issues/51744
transformers==5.14.1

# The vLLM version this whole setup (launch flags, env vars, throughput
# numbers in CLAUDE.md) was verified against. Not necessarily broken on a
# different 0.27.x point release, but untested - repin and re-verify
# end-to-end (see .claude/skills/run-doc2md/) before assuming a different
# version behaves the same.
vllm==0.27.1
